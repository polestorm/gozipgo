var ErrClosed = errors.New("zipreader: read after Close")

var ErrNoEntry = errors.New("zipreader: not an entry of this reader")

type Entry struct {
	Index int

	Name string

	IsDir bool

	Method uint16

	Flags uint16

	CRC32 uint32

	CompressedSize   uint64
	UncompressedSize uint64

	Modified Timestamp

	Comment string

	Extra []byte

	VersionMadeBy uint16
	VersionNeeded uint16
	InternalAttrs uint16
	ExternalAttrs uint32

	LocalHeaderOffset uint64

	DataOffset uint64

	DirectoryOffset int64

	Unsupported string
	// contains filtered or unexported fields
}

type Error struct {
	Code string

	Entry int

	Other int

	Name string

	Offset int64

	Message string
}

type NameEncoding int

type Options struct {
	MaxEntries int

	MaxDirectorySize int64

	MaxNameLength int

	MaxCommentLength int

	MaxExtraLength int

	MaxEntrySize int64

	MaxTotalSize int64

	MaxRatio int64

	AllowPrefix bool

	UnflaggedNames NameEncoding

	CodePage *[128]rune

	FoldCase bool
}

type Reader struct {
	// contains filtered or unexported fields
}

type Relaxation struct {
	Option string
	Entry  int
}

type Timestamp struct {
	Year, Month, Day int

	Hour, Minute, Second int

	Valid bool
}
