	Stored  uint16 = 0
	Deflate uint16 = 8
)

type Error struct {
	Code string

	Entry int

	Name string

	Message string
}

type Header struct {
	Name string

	Method uint16

	Modified Timestamp

	Zip64 bool
}

type RawInfo struct {
	CompressedSize   uint64
	UncompressedSize uint64
	CRC32            uint32
}

type Timestamp struct {
	Year, Month, Day     int
	Hour, Minute, Second int
}
